usage: sumbi compare [-h] [--home HOME] [--agents AGENTS] [--project PROJECT]
                     [--match-origin MATCH_ORIGIN] [--match-path MATCH_PATH]
                     [--idle-minutes IDLE_MINUTES] [--json OUT]
                     [--salt-file SALT_FILE]
                     [--outcome-source {github,local-verify}]
                     [--ledger LEDGER] [--outcomes github|DIR]
                     [--repository REPOSITORY] [--verify VERIFY]
                     [--scan-until SCAN_UNTIL]
                     [--active-minutes ACTIVE_MINUTES] [--cache DIR]
                     [--record DIR] --registration REGISTRATION
                     [--interventions FILE]
                     [--intervention-id INTERVENTION_ID] [--seed SEED]
                     [--resamples RESAMPLES]

options:
  -h, --help            show this help message and exit
  --home HOME
  --agents AGENTS
  --project PROJECT
  --match-origin MATCH_ORIGIN
  --match-path MATCH_PATH
  --idle-minutes IDLE_MINUTES
  --json OUT
  --salt-file SALT_FILE
                        Local pseudonym key file (overrides SUMBI_SALT)
  --outcome-source {github,local-verify}
                        Local verification reports verifier_never_passed with
                        exit-code counts; compare blocks before non-
                        inferiority
  --ledger LEDGER
  --outcomes github|DIR
  --repository REPOSITORY
                        Repository scope for local verification units
  --verify VERIFY       Repository-relative executed verification script
  --scan-until SCAN_UNTIL
                        Lifetime observation end for local verification
  --active-minutes ACTIVE_MINUTES
  --cache DIR
  --record DIR
  --registration REGISTRATION
  --interventions FILE  Optional install intervention JSONL; exclude and count
                        actual apply-time exposure gaps
  --intervention-id INTERVENTION_ID
                        Apply record ID (defaults to registration ID; must
                        match it)
  --seed SEED
  --resamples RESAMPLES
